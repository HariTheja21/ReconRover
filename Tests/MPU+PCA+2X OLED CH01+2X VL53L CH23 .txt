#include <Wire.h>
#include <Adafruit_GFX.h>
#include <Adafruit_SSD1306.h>
#include <Adafruit_MPU6050.h>
#include <Adafruit_Sensor.h>
#include <Adafruit_VL53L0X.h>

#define SDA_PIN 8
#define SCL_PIN 9

#define PCA9548A_ADDR 0x70
#define OLED_ADDR 0x3C

Adafruit_SSD1306 display(128, 64, &Wire, -1);
Adafruit_MPU6050 mpu;

Adafruit_VL53L0X vl53Front;
Adafruit_VL53L0X vl53Scan;

void selectChannel(uint8_t channel) {
  Wire.beginTransmission(PCA9548A_ADDR);
  Wire.write(1 << channel);
  Wire.endTransmission();
}

uint16_t readVL53(uint8_t channel, Adafruit_VL53L0X &sensor) {

  VL53L0X_RangingMeasurementData_t measure;

  selectChannel(channel);

  sensor.rangingTest(&measure, false);

  return measure.RangeMilliMeter;
}

void setup() {

  Serial.begin(115200);
  Wire.begin(SDA_PIN, SCL_PIN);

  Serial.println("Recon Rover Full Sensor Test");

  if (!mpu.begin()) {
    Serial.println("MPU6050 FAIL");
    while (1);
  }

  Serial.println("MPU6050 OK");

  selectChannel(2);
  if (!vl53Front.begin()) {
    Serial.println("Front VL53 FAIL");
    while (1);
  }

  Serial.println("Front VL53 OK");

  selectChannel(3);
  if (!vl53Scan.begin()) {
    Serial.println("Scan VL53 FAIL");
    while (1);
  }

  Serial.println("Scan VL53 OK");

  selectChannel(0);
  display.begin(SSD1306_SWITCHCAPVCC, OLED_ADDR);

  selectChannel(1);
  display.begin(SSD1306_SWITCHCAPVCC, OLED_ADDR);

  Serial.println("SYSTEM READY");
}

void loop() {

  sensors_event_t a, g, temp;
  mpu.getEvent(&a, &g, &temp);

  uint16_t frontDistance = readVL53(2, vl53Front);
  uint16_t scanDistance  = readVL53(3, vl53Scan);

  Serial.print("Front: ");
  Serial.print(frontDistance);
  Serial.print(" mm");

  Serial.print(" | Scan: ");
  Serial.print(scanDistance);
  Serial.print(" mm");

  Serial.print(" | AX:");
  Serial.print(a.acceleration.x);

  Serial.print(" AY:");
  Serial.print(a.acceleration.y);

  Serial.print(" AZ:");
  Serial.println(a.acceleration.z);

  // LEFT OLED
  selectChannel(0);

  display.clearDisplay();

  display.setTextSize(1);
  display.setTextColor(SSD1306_WHITE);

  display.setCursor(0, 0);
  display.println("FRONT");

  display.setTextSize(2);
  display.setCursor(0, 20);
  display.print(frontDistance);
  display.println("mm");

  display.display();

  // RIGHT OLED
  selectChannel(1);

  display.clearDisplay();

  display.setTextSize(1);
  display.setTextColor(SSD1306_WHITE);

  display.setCursor(0, 0);
  display.println("SCAN");

  display.setTextSize(2);
  display.setCursor(0, 20);
  display.print(scanDistance);
  display.println("mm");

  display.display();

  delay(100);
}
